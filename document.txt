app = Flask(__name__)

@app.route("/")
def home():

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=5000)

Dockerfile
# Base image
FROM python:3.10-slim

# Set work directory
WORKDIR /app

# Copy dependencies
COPY requirements.txt .

# Install packages
RUN pip install --no-cache-dir -r requirements.txt

# Copy the application
COPY . .

# Expose port
EXPOSE 5000

# Run application
CMD ["python", "app.py"]

docker build -t srika-webapp:v1 .
